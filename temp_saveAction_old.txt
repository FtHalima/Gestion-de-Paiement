@FXML
    private void saveAction() {
        if (!validateForm()) {
            return;
        }

        // Prepare professeur
        Professeur professeur;
        if (currentProfesseur != null && !isNewProfessorMode) {
            professeur = currentProfesseur;
            // Update mutable fields from the form (fields that are editable in display mode?)
            // In display mode, only nom, prenom, ddr, grade, echelle, affectation are shown as labels; we don't have edit fields.
            // Since we are not in edit mode for existing professor, we assume user cannot change them here.
            // However, if they changed via some other means, we would need to update; but not required.
            // We'll still update the fields that are always present in the form (RIB fields) only if we are in new professor mode.
            // For safety, we will copy over the editable fields from the form fields if they are not null.
            // But the edit fields are hidden when not in new professor mode; their values may be stale.
            // So we only update from the form when in new professor mode.
            // For existing professor, we rely on the data already loaded from DB.
            // No changes needed.
        } else {
            professeur = new Professeur();
            professeur.setCin(cinField.getText().trim());
            professeur.setPpr(pprField.getText().trim());
            professeur.setNom(nomField.getText().trim());
            professeur.setPrenom(prenomField.getText().trim());
            professeur.setDdr(ddrPicker.getValue());
            professeur.setGrade(gradeField.getValue());
            professeur.setEchelle(echelleField.getText().trim().isEmpty() ? null : Integer.valueOf(echelleField.getText().trim()));
            professeur.setAffectation(affectationField.getText().trim());
            // Set RIB fields from the form
            professeur.setRibBanque(banqueField.getText());
            professeur.setRibVille(villeField.getText());
            professeur.setRibNumeroCompte(numeroCompteField.getText());
            professeur.setRibCle(cleField.getText());
            // Log before saving
            System.out.println("RIB AVANT ENREGISTREMENT : banque=" + professeur.getRibBanque()
                    + ", ville=" + professeur.getRibVille()
                    + ", numero=" + professeur.getRibNumeroCompte()
                    + ", cle=" + professeur.getRibCle());
        }

        // Create paiement
        Paiement paiement = new Paiement();
        paiement.setProfesseur(professeur);
        paiement.setTypePaiement(typePaiementCombo.getSelectionModel().getSelectedItem());
        // Objet et référence règlement : convert "-" to null for cleaner DB
        String objetReglement = objetReglementLabel.getText();
        paiement.setObjetReglement("-".equals(objetReglement) || objetReglement.isEmpty() ? null : objetReglement);
        String referenceReglement = referenceReglementLabel.getText();
        paiement.setReferenceReglement("-".equals(referenceReglement) || referenceReglement.isEmpty() ? null : referenceReglement);
        paiement.setDateDebut(dateDebutField.getValue());
        paiement.setDateFin(dateFinField.getValue());
        paiement.setNombreHeures(parseBigDecimal(nombreHeuresField.getText()));
        paiement.setTaux(parseBigDecimal(tauxField.getText()));
        paiement.setTauxIr(parseBigDecimal(irCombo.getSelectionModel().getSelectedItem()));
        paiement.setModePaiement(modePaiementLabel.getText());
        paiement.setTypeReferenceReglement(typeReferenceReglementLabel.getText());
        paiement.setDatePaiement(datePaiementField.getValue());

        // Save via service
        Paiement saved = paiementService.enregistrerPaiement(paiement);

        // Log after saving (professor should have ID if new)
        if (professeur.getIdProfesseur() != null) {
            System.out.println("RIB APRÈS ENREGISTREMENT : idProfesseur=" + professeur.getIdProfesseur()
                    + ", banque=" + professeur.getRibBanque()
                    + ", ville=" + professeur.getRibVille()
                    + ", numero=" + professeur.getRibNumeroCompte()
                    + ", cle=" + professeur.getRibCle());
        } else {
            System.out.println("RIB APRÈS ENREGISTREMENT : professeur ID null (maybe existing)");
        }

        // Show success
        Alert successAlert = new Alert(Alert.AlertType.INFORMATION);
        successAlert.setTitle("Paiement enregistré");
        successAlert.setHeaderText(null);
        successAlert.setContentText("Paiement enregistré avec succès. ID: " + saved.getIdPaiement());
        successAlert.showAndWait();

        // Optionally reset form
        resetAction();
    }
